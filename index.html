/** * Pere Kaaluralli Pilve-Andmebaas (Google Sheets versioon) * See skript võimaldab HTML rakendusel salvestada ja lugeda andmeid tabelist. */ const SHEET_NAME = 'data'; // Tabeli lehe nimi /** * Teeb kindlaks, et vajalik tabelileht on olemas. */ function getSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(SHEET_NAME); if (!sh) { sh = ss.insertSheet(SHEET_NAME); } // Kui tabel on tühi, lisame päised if (sh.getLastRow() === 0) { sh.appendRow(['raceId', 'stateJson', 'updatedAt']); } return sh; } /** * Otsib tabelist rida vastavalt võistluse ID-le. */ function findRowByRaceId_(sh, raceId) { const last = sh.getLastRow(); if (last < 2) return -1; const values = sh.getRange(2, 1, last - 1, 1).getValues(); for (let i = 0; i < values.length; i++) { if (String(values[i][0]) === String(raceId)) { return i + 2; // Tagastab rea numbri (indeks + 2, sest päis on esimene rida) } } return -1; } /** * GET päring: Andmete lugemine. * Kasutus: URL?raceId=pere-voidujooks */ function doGet(e) { try { const raceId = (e && e.parameter && e.parameter.raceId) ? String(e.parameter.raceId).trim() : ''; if (!raceId) return createResponse_({ ok: false, error: 'raceId puudub' }); const sh = getSheet_(); const row = findRowByRaceId_(sh, raceId); if (row === -1) { return createResponse_({ ok: true, found: false, state: null }); } const stateJson = sh.getRange(row, 2).getValue(); const updatedAt = sh.getRange(row, 3).getValue(); let state = null; try { state = JSON.parse(stateJson); } catch (err) { return createResponse_({ ok: false, error: 'Vigane JSON tabelis' }); } return createResponse_({ ok: true, found: true, state, updatedAt }); } catch (ex) { return createResponse_({ ok: false, error: String(ex) }); } } /** * POST päring: Andmete salvestamine. * Võtab vastu JSON paketi: { raceId, state } */ function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { return createResponse_({ ok: false, error: 'Andmed puuduvad' }); } const body = JSON.parse(e.postData.contents); const raceId = body && body.raceId ? String(body.raceId).trim() : ''; const state = body && body.state ? body.state : null; if (!raceId) return createResponse_({ ok: false, error: 'raceId puudub' }); if (!state) return createResponse_({ ok: false, error: 'state puudub' }); // MÄRKUS: Tokeni kontroll on eemaldatud kasutaja soovil mugavuse huvides. const sh = getSheet_(); const row = findRowByRaceId_(sh, raceId); const now = new Date(); const jsonString = JSON.stringify(state); if (row === -1) { // Lisame uue rea sh.appendRow([raceId, jsonString, now]); } else { // Uuendame olemasolevat rida sh.getRange(row, 2).setValue(jsonString); sh.getRange(row, 3).setValue(now); } return createResponse_({ ok: true, saved: true, updatedAt: now }); } catch (ex) { return createResponse_({ ok: false, error: String(ex) }); } } /** * Abifunktsioon JSON vastuse loomiseks. */ function createResponse_(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
